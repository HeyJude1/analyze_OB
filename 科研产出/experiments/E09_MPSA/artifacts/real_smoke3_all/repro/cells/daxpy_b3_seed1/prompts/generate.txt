int daxpy_k(BLASLONG n, BLASLONG dummy0, BLASLONG dummy1, double da, double *x, BLASLONG inc_x, double *y, BLASLONG inc_y, double *dummy, BLASLONG dummy2)
{
 BLASLONG i=0;
 BLASLONG ix,iy;

 if ( n < 0 ) return(0);
 if ( da == 0.0 ) return(0);

 ix = 0;
 iy = 0;

 while(i < n)
 {

  y[iy] += da * x[ix] ;
  ix += inc_x ;
  iy += inc_y ;
  i++ ;

 }
 return(0);

  `vfloat64m1_t vred = __riscv_vfredusum_vs_f64m8_f64(vsum, __riscv_vfmv_v_f_f64m1(0.0, 1), vl);`
